// Finds an existing Material List entry (case-insensitive, trimmed // match) for a purchase row's Material text, or creates a new one // seeded from that row's Material/Supplier/Unit. When a matching // material already exists, its Supplier is kept in sync with whatever // this purchase row has — so a supplier entered/updated in Purchase // History flows through to the Material List, and from there to // Inventory (which reads its Supplier column straight from Material // List). Returns the matching material row, or null if the purchase // row has no material name yet. async function ensureMaterialForPurchase(row){ const name = (row.material || '').trim(); if(!name) return null; const existing = findMaterialByName(name); const supplier = (row.supplier || '').trim(); if(existing){ if(supplier && existing.supplier !== supplier){ existing.supplier = supplier; await persistMaterialRow(existing); renderMaterialRows(); renderInventoryRows(); } return existing; } const created = { id: materialRowId(), material: name, supplier: row.supplier || '', brand: '', unit: row.unit || '', createdAt: new Date().toISOString() }; cachedMaterialRows.push(created); refreshMaterialsDatalist(); await persistMaterialRow(created); return created; } // Keeps exactly one Stock In row on the ledger for this purchase row // (id `purchase-`), so editing a purchase's Material/Qty // updates Inventory's Current Stock instead of double-counting. async function syncPurchaseLedger(row){ const ledgerId = `purchase-${row.id}`; const material = await ensureMaterialForPurchase(row); if(!material){ cachedStockRows = cachedStockRows.filter(r => r.id !== ledgerId); try{ await fetch(`${STOCK_API}&id=${encodeURIComponent(ledgerId)}`, { method: 'DELETE', headers: { ...authHeaders() } }); }catch(e){ /* best effort */ } return; } const entry = { id: ledgerId, date: row.date || '', materialId: material.id, material: material.material, type: 'in', qty: parseFloat(row.qty) || 0, unit: row.unit || material.unit || '', source: 'purchase', reference: [row.poNumber, row.drNumber].filter(Boolean).join(' / ') || 'Purchase', personnel: row.preparedBy || '', notes: row.memo || '', createdAt: row.createdAt || new Date().toISOString() }; const idx = cachedStockRows.findIndex(r => r.id === ledgerId); if(idx >= 0) cachedStockRows[idx] = entry; else cachedStockRows.push(entry); await persistStockRow(entry); } // Autocomplete/suggestion handler for the Material field: if the // typed name matches an existing Material List entry, reuse it and // fill in its Supplier/Unit (when this row hasn't set its own yet) // instead of letting a near-duplicate get created. This only saves // the purchase row itself — it does NOT touch the Inventory ledger. // That only happens when the row's own Save button is clicked (see // saveRowToInventory), so Current Stock never updates from a row // that's still being typed into. function onPurchaseMaterialChange(el){ const id = el.dataset.id; const row = cachedPurchaseRows.find(r => r.id === id); if(!row) return; row.material = el.value; const match = findMaterialByName(row.material); if(match){ if(!row.supplier) row.supplier = match.supplier || ''; if(!row.unit) row.unit = match.unit || ''; } renderPurchaseRows(); persistPurchaseRow(row); } // Saves the field's own value only — does NOT sync to Inventory. // Inventory sync is explicit now, via each row's Save button. function savePurchaseField(el){ const id = el.dataset.id; const field = el.dataset.field; const value = el.value; const row = cachedPurchaseRows.find(r => r.id === id); if(!row) return; row[field] = value; persistPurchaseRow(row); } // The only path that updates Inventory's Current Stock from a // purchase row. Clicking a row's Save button persists that row and // then creates/updates its linked Stock In entry — so editing a // purchase mid-typing never touches Inventory until you explicitly // confirm the row with Save. async function saveRowToInventory(id){ const row = cachedPurchaseRows.find(r => r.id === id); if(!row) return; if(!(row.material || '').trim()){ showToast('Add a material name to this row before saving.', 'error'); return; } await commitPoNumber(row); renderPurchaseRows(); await persistPurchaseRow(row); await syncPurchaseLedger(row); showToast('Saved — synced to Inventory.', 'success'); }
